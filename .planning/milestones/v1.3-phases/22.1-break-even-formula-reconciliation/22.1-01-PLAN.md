Depreciation (mirror calculateDepreciation, costCalc.ts:70-80):
  purchasePrice    = printerInstance.actualPurchasePrice ?? printerConfig.purchasePrice ?? 0
  recoveryMonths   = printerInstance.recoveryMonths ?? 12
  monthlyHours     = printerInstance.estimatedMonthlyPrintHours ?? 40
  totalRecoveryHrs = recoveryMonths * monthlyHours
  perHour          = totalRecoveryHrs > 0 ? purchasePrice / totalRecoveryHrs : 0
  depreciation     = perHour * job.printTimeHours
  (no failureRate multiplier)

Nozzle wear (mirror calculateNozzleWear, costCalc.ts:86-99):
  totalVolumeCm3 = Σ over job.filaments of:
      grams / getMaterialDensity(materials.find(m => m.id === filament.filamentId)?.filamentType ?? null)
    (filaments with grams <= 0 contribute 0)
  nozzleWear = printerConfig
    ? (totalVolumeCm3 / printerConfig.nozzleLifespanCm3) * printerConfig.nozzleCost
    : 0
  (no failureRate multiplier)
</formulas>
